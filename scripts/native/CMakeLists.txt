cmake_minimum_required(VERSION 3.15)
project(sirui_dashboard_tools LANGUAGES CXX)
add_library(workspace_scan STATIC workspace_scan.cpp)
target_compile_features(workspace_scan PUBLIC cxx_std_17)
add_executable(workspace_inventory workspace_inventory.cpp)
target_link_libraries(workspace_inventory PRIVATE workspace_scan)
target_compile_features(workspace_inventory PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(workspace_inventory PRIVATE /utf-8 /W4)
  target_compile_options(workspace_scan PRIVATE /utf-8 /W4)
else()
  target_compile_options(workspace_inventory PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS workspace_inventory RUNTIME DESTINATION bin)
enable_testing()
add_executable(workspace_inventory_test workspace_inventory_test.cpp)
target_link_libraries(workspace_inventory_test PRIVATE workspace_scan)
if(MSVC)
  target_compile_options(workspace_inventory_test PRIVATE /utf-8)
endif()
add_test(NAME workspace_files COMMAND workspace_inventory_test)
