cmake_minimum_required(VERSION 3.15)
project(sirui_dashboard_tools LANGUAGES CXX)
add_library(workspace_scan STATIC workspace_scan.cpp)
target_compile_features(workspace_scan PUBLIC cxx_std_17)
add_executable(nav_scrollspy nav_scrollspy.cpp)
target_compile_features(nav_scrollspy PRIVATE cxx_std_17)
add_executable(workspace_inventory workspace_inventory.cpp)
target_link_libraries(workspace_inventory PRIVATE workspace_scan)
target_compile_features(workspace_inventory PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(workspace_inventory PRIVATE /utf-8 /W4)
  target_compile_options(workspace_scan PRIVATE /utf-8 /W4)
  target_compile_options(nav_scrollspy PRIVATE /utf-8 /W4)
else()
  target_compile_options(workspace_inventory PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(nav_scrollspy PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS workspace_inventory nav_scrollspy RUNTIME DESTINATION bin)
set(template_dir "${CMAKE_CURRENT_SOURCE_DIR}/../templates")
set(render_sources "${CMAKE_CURRENT_SOURCE_DIR}/page_renderer.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/page_renderer.hpp" "${CMAKE_CURRENT_SOURCE_DIR}/compiled_pages.hpp.in"
    "${template_dir}/pagefind.html" "${template_dir}/daily-download.html")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${render_sources})
set(render_hashes "")
foreach(source IN LISTS render_sources)
  file(SHA256 "${source}" digest)
  string(APPEND render_hashes "${digest}")
endforeach()
string(SHA256 PAGE_ID "${render_hashes}")
file(READ "${template_dir}/pagefind.html" INDEX_PAGE)
file(READ "${template_dir}/daily-download.html" DAILY_PAGE)
string(REPLACE "\r\n" "\n" INDEX_PAGE "${INDEX_PAGE}")
string(REPLACE "\r\n" "\n" DAILY_PAGE "${DAILY_PAGE}")
configure_file(compiled_pages.hpp.in compiled_pages.hpp @ONLY NEWLINE_STYLE UNIX)
add_library(page_objects OBJECT page_renderer.cpp)
target_compile_features(page_objects PUBLIC cxx_std_17)
target_include_directories(page_objects PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
set_target_properties(page_objects PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(page_renderer_native SHARED $<TARGET_OBJECTS:page_objects>)
add_executable(page_renderer page_renderer_cli.cpp $<TARGET_OBJECTS:page_objects>)
target_compile_features(page_renderer PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(page_objects PRIVATE /utf-8 /W4)
  target_compile_options(page_renderer PRIVATE /utf-8 /W4)
endif()
install(TARGETS page_renderer page_renderer_native RUNTIME DESTINATION bin LIBRARY DESTINATION bin)
enable_testing()
add_test(NAME rendered_pages COMMAND ${CMAKE_COMMAND} -DTOOL=$<TARGET_FILE:page_renderer>
  -P ${CMAKE_CURRENT_SOURCE_DIR}/test_page_renderer.cmake)
add_executable(workspace_inventory_test workspace_inventory_test.cpp)
target_link_libraries(workspace_inventory_test PRIVATE workspace_scan)
if(MSVC)
  target_compile_options(workspace_inventory_test PRIVATE /utf-8)
endif()
add_test(NAME workspace_files COMMAND workspace_inventory_test)
add_test(NAME nav_scrollspy_pages COMMAND ${CMAKE_COMMAND} -DTOOL=$<TARGET_FILE:nav_scrollspy>
  -DPROJECT_ROOT=${CMAKE_CURRENT_SOURCE_DIR}/../.. -P ${CMAKE_CURRENT_SOURCE_DIR}/test_nav_scrollspy.cmake)
